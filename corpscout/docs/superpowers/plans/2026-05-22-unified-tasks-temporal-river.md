# Unified Tasks: Temporal + River Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the local Temporal execution mirror with direct Temporal workflow starts and a unified Tasks API that combines Temporal visibility with River jobs.

**Architecture:** Add a focused `internal/tasksvc` package that owns Temporal workflow metadata, direct Temporal starts, Temporal visibility reads, and normalized task response shapes. River task list/stat projections are generated by sqlc from `database/queries/tasks.sql`; Temporal state is read through the Temporal SDK. The UI moves from split River/Temporal tabs to one task table backed by `/api/v1/tasks`.

**Tech Stack:** Go, chi, pgx, sqlc, PostgreSQL migrations, River, Temporal Go SDK, React Router, TypeScript, shadcn/ui.

-- name: ListRiverTasks :many
WITH normalized AS (
    SELECT
        format('river:%s', j.id)::text AS id,
        'river'::text AS executor,
        COALESCE(CASE
            WHEN j.kind IN ('source_pull', 'source_process') THEN j.args->>'source_name'
            WHEN j.kind = 'enrich_company_financials' THEN j.args->>'source_name'
            ELSE NULL
        END, '')::text AS source,
        CASE
            WHEN j.kind = 'source_pull' THEN 'download'
            WHEN j.kind = 'source_process' THEN 'process'
            WHEN j.kind = 'domain_resolve' THEN 'domain_resolve'
            WHEN j.kind = 'domain_import' THEN 'domain_import'
            WHEN j.kind = 'enrich_company_financials' THEN 'enrich_company_financials'
            ELSE j.kind
        END::text AS action,
        j.kind::text AS kind,
        CASE
            WHEN j.state::text IN ('pending', 'available', 'scheduled', 'retryable') THEN 'queued'
            WHEN j.state::text = 'running' THEN 'running'
            WHEN j.state::text = 'completed' THEN 'completed'
            WHEN j.state::text = 'discarded' THEN 'failed'
            WHEN j.state::text = 'cancelled' THEN 'cancelled'
            ELSE j.state::text
        END::text AS status,
        j.state::text AS native_status,
        NULL::text AS workflow_id,
        NULL::text AS workflow_run_id,
        j.id AS river_job_id,
        COALESCE(CASE
            WHEN j.kind IN ('source_pull', 'source_process') THEN j.args->>'source_name'
            WHEN j.kind IN ('domain_resolve', 'enrich_company_financials') THEN COALESCE(c.name, j.args->>'company_id')
            ELSE NULL
        END, '')::text AS subject,
        j.created_at,
        COALESCE(j.attempted_at, j.scheduled_at, j.created_at) AS started_at,
        COALESCE(to_char(j.finalized_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"'), '')::text AS closed_at,
        j.args,
        array_to_json(j.errors)::jsonb AS errors,
        j.attempt::int AS attempt,
        j.max_attempts::int AS max_attempts,
        j.queue::text AS queue,
        j.priority::int AS priority,
        j.scheduled_at,
        COALESCE(CASE
            WHEN array_length(j.errors, 1) > 0
                THEN j.errors[array_length(j.errors, 1)]->>'error'
        END, '')::text AS last_error
    FROM river_job j
    LEFT JOIN companies c
        ON j.kind IN ('domain_resolve', 'enrich_company_financials')
       AND c.id = (j.args->>'company_id')::uuid
)
SELECT *
FROM normalized
WHERE (sqlc.narg('source')::text IS NULL OR source = sqlc.narg('source'))
  AND (sqlc.narg('action')::text IS NULL OR action = sqlc.narg('action'))
  AND (sqlc.narg('status')::text IS NULL OR status = sqlc.narg('status'))
  AND (sqlc.narg('kind')::text IS NULL OR kind = sqlc.narg('kind'))
  AND (sqlc.narg('before')::timestamptz IS NULL OR created_at < sqlc.narg('before'))
ORDER BY created_at DESC
LIMIT sqlc.arg('limit') OFFSET 0;

-- name: ListRiverTaskStats :many
WITH normalized AS (
    SELECT
        'river'::text AS executor,
        COALESCE(CASE
            WHEN j.kind IN ('source_pull', 'source_process') THEN j.args->>'source_name'
            WHEN j.kind = 'enrich_company_financials' THEN j.args->>'source_name'
            ELSE NULL
        END, '')::text AS source,
        CASE
            WHEN j.kind = 'source_pull' THEN 'download'
            WHEN j.kind = 'source_process' THEN 'process'
            WHEN j.kind = 'domain_resolve' THEN 'domain_resolve'
            WHEN j.kind = 'domain_import' THEN 'domain_import'
            WHEN j.kind = 'enrich_company_financials' THEN 'enrich_company_financials'
            ELSE j.kind
        END::text AS action,
        CASE
            WHEN j.state::text IN ('pending', 'available', 'scheduled', 'retryable') THEN 'queued'
            WHEN j.state::text = 'running' THEN 'running'
            WHEN j.state::text = 'completed' THEN 'completed'
            WHEN j.state::text = 'discarded' THEN 'failed'
            WHEN j.state::text = 'cancelled' THEN 'cancelled'
            ELSE j.state::text
        END::text AS status
    FROM river_job j
)
SELECT executor, COALESCE(source, '')::text AS source, action, status, COUNT(*)::bigint AS count
FROM normalized
WHERE (sqlc.narg('source')::text IS NULL OR source = sqlc.narg('source'))
  AND (sqlc.narg('action')::text IS NULL OR action = sqlc.narg('action'))
  AND (sqlc.narg('status')::text IS NULL OR status = sqlc.narg('status'))
GROUP BY executor, COALESCE(source, ''), action, status
ORDER BY executor, source, action, status;
```

package tasksvc

import (
	"context"
	"fmt"
	"sort"
	"strings"
	"time"

	"github.com/cockroachdb/errors"
	enumspb "go.temporal.io/api/enums/v1"
	commonpb "go.temporal.io/api/common/v1"
	workflowpb "go.temporal.io/api/workflow/v1"
	"go.temporal.io/api/workflowservice/v1"
	"go.temporal.io/sdk/client"
	"go.temporal.io/sdk/converter"
)

const (
	Namespace             = "corpscout"
	TaskQueue             = "corpscout-pipelines"
	SearchAttributeSource = "CorpscoutSource"
	SearchAttributeAction = "CorpscoutAction"
)

var knownTemporalActions = []string{"download", "translate", "create_suggestions"}
var knownTemporalSources = []string{"brreg", "cvr", "gleif", "ariregister", "companies_house"}

type TemporalClient interface {
	ExecuteWorkflow(context.Context, client.StartWorkflowOptions, interface{}, ...interface{}) (client.WorkflowRun, error)
	ListWorkflow(context.Context, *workflowservice.ListWorkflowExecutionsRequest) (*workflowservice.ListWorkflowExecutionsResponse, error)
	CountWorkflow(context.Context, *workflowservice.CountWorkflowExecutionsRequest) (*workflowservice.CountWorkflowExecutionsResponse, error)
	GetSearchAttributes(context.Context) (*workflowservice.GetSearchAttributesResponse, error)
}

func ValidateTemporalSearchAttributes(ctx context.Context, tc TemporalClient) error {
	resp, err := tc.GetSearchAttributes(ctx)
	if err != nil {
		return errors.Wrap(err, "get temporal search attributes")
	}
	required := map[string]enumspb.IndexedValueType{
		SearchAttributeSource: enumspb.INDEXED_VALUE_TYPE_KEYWORD,
		SearchAttributeAction: enumspb.INDEXED_VALUE_TYPE_KEYWORD,
	}
	var missing []string
	for name, expected := range required {
		if resp.GetKeys()[name] != expected {
			missing = append(missing, name)
		}
	}
	sort.Strings(missing)
	if len(missing) > 0 {
		return errors.Newf("missing temporal search attributes in namespace %s: %s", Namespace, strings.Join(missing, ", "))
	}
	return nil
}

func searchAttributes(source, action string) map[string]interface{} {
	return map[string]interface{}{
		SearchAttributeSource: source,
		SearchAttributeAction: action,
	}
}

func memo(source, action, trigger string, scope map[string]any) map[string]interface{} {
	return map[string]interface{}{
		"source":  source,
		"action":  action,
		"trigger": trigger,
		"scope":   scope,
	}
}

func buildTemporalVisibilityQuery(filter TaskFilter) string {
	parts := make([]string, 0, 5)
	if filter.Source != "" {
		parts = append(parts, fmt.Sprintf("%s = '%s'", SearchAttributeSource, escapeVisibilityValue(filter.Source)))
	}
	if filter.Action != "" {
		parts = append(parts, fmt.Sprintf("%s = '%s'", SearchAttributeAction, escapeVisibilityValue(filter.Action)))
	} else {
		quoted := make([]string, 0, len(knownTemporalActions))
		for _, action := range knownTemporalActions {
			quoted = append(quoted, fmt.Sprintf("'%s'", escapeVisibilityValue(action)))
		}
		parts = append(parts, fmt.Sprintf("%s IN (%s)", SearchAttributeAction, strings.Join(quoted, ", ")))
	}
	if filter.Kind != "" {
		parts = append(parts, fmt.Sprintf("WorkflowType = '%s'", escapeVisibilityValue(filter.Kind)))
	}
	if filter.Status != "" {
		parts = append(parts, temporalStatusQuery(filter.Status))
	}
	if filter.Before != nil {
		parts = append(parts, fmt.Sprintf("StartTime < '%s'", filter.Before.UTC().Format(time.RFC3339)))
	}
	return strings.Join(parts, " AND ")
}

func temporalStatusQuery(status string) string {
	switch status {
	case "running":
		return "ExecutionStatus = 'Running'"
	case "completed":
		return "ExecutionStatus = 'Completed'"
	case "failed":
		return "ExecutionStatus IN ('Failed', 'TimedOut', 'Terminated')"
	case "cancelled":
		return "ExecutionStatus = 'Canceled'"
	case "queued":
		return "ExecutionStatus = 'Running'"
	default:
		return fmt.Sprintf("ExecutionStatus = '%s'", escapeVisibilityValue(status))
	}
}

func escapeVisibilityValue(value string) string {
	return strings.ReplaceAll(value, "'", "''")
}

func temporalStatus(status enumspb.WorkflowExecutionStatus) (string, string) {
	switch status {
	case enumspb.WORKFLOW_EXECUTION_STATUS_RUNNING:
		return "running", "Running"
	case enumspb.WORKFLOW_EXECUTION_STATUS_COMPLETED:
		return "completed", "Completed"
	case enumspb.WORKFLOW_EXECUTION_STATUS_FAILED:
		return "failed", "Failed"
	case enumspb.WORKFLOW_EXECUTION_STATUS_TIMED_OUT:
		return "failed", "TimedOut"
	case enumspb.WORKFLOW_EXECUTION_STATUS_TERMINATED:
		return "failed", "Terminated"
	case enumspb.WORKFLOW_EXECUTION_STATUS_CANCELED:
		return "cancelled", "Canceled"
	default:
		return "unknown", status.String()
	}
}

func temporalTaskFromExecution(info *workflowpb.WorkflowExecutionInfo, temporalUIURL string) TaskItem {
	status, nativeStatus := temporalStatus(info.GetStatus())
	source := searchAttributeString(info.GetSearchAttributes(), SearchAttributeSource)
	action := searchAttributeString(info.GetSearchAttributes(), SearchAttributeAction)
	workflowID := info.GetExecution().GetWorkflowId()
	runID := info.GetExecution().GetRunId()
	started := info.GetStartTime().AsTime()
	var closed *time.Time
	if info.GetCloseTime() != nil {
		t := info.GetCloseTime().AsTime()
		closed = &t
	}
	item := TaskItem{
		ID:            fmt.Sprintf("temporal:%s:%s", workflowID, runID),
		Executor:      ExecutorTemporal,
		Source:        source,
		Action:        action,
		Kind:          info.GetType().GetName(),
		Status:        status,
		NativeStatus:  nativeStatus,
		WorkflowID:    workflowID,
		WorkflowRunID: runID,
		Subject:       source,
		CreatedAt:     started,
		StartedAt:     started,
		ClosedAt:      closed,
	}
	if temporalUIURL != "" {
		item.ExternalURL = fmt.Sprintf("%s/namespaces/%s/workflows/%s/%s", strings.TrimRight(temporalUIURL, "/"), Namespace, workflowID, runID)
	}
	return item
}

func searchAttributeString(attrs *commonpb.SearchAttributes, key string) string {
	if attrs == nil || attrs.IndexedFields == nil {
		return ""
	}
	payload := attrs.IndexedFields[key]
	if payload == nil {
		return ""
	}
	var value string
	if err := converter.GetDefaultDataConverter().FromPayload(payload, &value); err != nil {
		return ""
	}
	return value
}

package tasksvc

import "strings"

type SourceWorkflow struct {
	WorkflowType string
	Country      string
	FirstMode    string
	NextMode     string
	BulkFirst    bool
}

var sourceWorkflows = map[string]SourceWorkflow{
	"companies_house": {WorkflowType: "PullCompaniesHouse", Country: "GB"},
	"brreg":           {WorkflowType: "PullBrreg", Country: "NO", FirstMode: "bulk", NextMode: "incremental", BulkFirst: true},
	"gleif":           {WorkflowType: "PullGLEIF", FirstMode: "bulk", NextMode: "delta", BulkFirst: true},
	"cvr":             {WorkflowType: "PullCVR", Country: "DK", FirstMode: "bulk", NextMode: "incremental", BulkFirst: true},
	"ariregister":     {WorkflowType: "PullAriregister", Country: "EE", FirstMode: "bulk", NextMode: "refresh", BulkFirst: true},
}

func TemporalWorkflowForSource(source string) (SourceWorkflow, bool) {
	cfg, ok := sourceWorkflows[source]
	return cfg, ok
}

func sourceMode(cfg SourceWorkflow, checkpointExists bool, savedCursor string, source string) (mode string, incrementalFrom string) {
	if cfg.BulkFirst {
		mode = cfg.FirstMode
		if checkpointExists {
			mode = cfg.NextMode
		}
	}
	if source == "brreg" && strings.HasPrefix(savedCursor, "bulk:") {
		incrementalFrom = strings.TrimPrefix(savedCursor, "bulk:") + ",0"
	}
	return mode, incrementalFrom
}
```

package tasksvc

import (
	"context"
	"fmt"
	"os"
	"time"

	"github.com/cockroachdb/errors"
	enumspb "go.temporal.io/api/enums/v1"
	"go.temporal.io/sdk/client"

	db "github.com/pulsarpoint/corpscout/scheduler/internal/db/gen"
)

type Service struct {
	db            db.Querier
	temporal      TemporalClient
	temporalUIURL string
	now           func() time.Time
}

func NewService(q db.Querier, tc TemporalClient, temporalUIURL string) *Service {
	return &Service{db: q, temporal: tc, temporalUIURL: temporalUIURL, now: time.Now}
}

type StartResult struct {
	Executor      string `json:"executor"`
	Status        string `json:"status"`
	WorkflowID    string `json:"workflow_id,omitempty"`
	WorkflowRunID string `json:"workflow_run_id,omitempty"`
	RiverJobID     *int64 `json:"river_job_id,omitempty"`
}

type StartSourceDownloadRequest struct {
	Source  string
	Trigger string
	IDs     []string
	Force   bool
}

type StartTranslationRequest struct {
	Source       string
	WorkflowType string
	Trigger      string
	IDs          []string
	Filters      map[string]string
	SourceLang   string
	TableName     string
	FXRateDate   string
}

func (s *Service) StartSourceDownload(ctx context.Context, req StartSourceDownloadRequest) (StartResult, error) {
	if s.temporal == nil {
		return StartResult{}, errors.New("temporal client not available")
	}
	cfg, ok := TemporalWorkflowForSource(req.Source)
	if !ok {
		return StartResult{}, errors.Newf("no temporal workflow registered for source %q", req.Source)
	}

	savedCursor := ""
	checkpointExists := false
	if checkpoint, err := s.db.GetSyncCheckpoint(ctx, req.Source); err == nil {
		checkpointExists = true
		savedCursor = checkpoint.Cursor
	}
	mode, incrementalFrom := sourceMode(cfg, checkpointExists, savedCursor, req.Source)
	country := cfg.Country
	workflowID := sourceWorkflowID(req.Source, country, s.now())

	if err := s.db.UpdateSourcePullStarted(ctx, req.Source); err != nil {
		return StartResult{}, errors.Wrap(err, "update source pull started")
	}

	run, err := s.temporal.ExecuteWorkflow(ctx, client.StartWorkflowOptions{
		ID:        workflowID,
		TaskQueue: TaskQueue,
		SearchAttributes: searchAttributes(req.Source, "download"),
		Memo: memo(req.Source, "download", req.Trigger, map[string]any{
			"ids":              req.IDs,
			"force":            req.Force,
			"mode":             mode,
			"cursor":           savedCursor,
			"incremental_from": incrementalFrom,
		}),
	}, cfg.WorkflowType, map[string]any{
		"country":          country,
		"ids":              req.IDs,
		"corpscout_run_id": workflowID,
		"run_id":           workflowID,
		"cursor":           savedCursor,
		"mode":             mode,
		"incremental_from": incrementalFrom,
		"force":            req.Force,
	})
	if err != nil {
		return StartResult{}, errors.Wrap(err, "start temporal source workflow")
	}

	return StartResult{Executor: ExecutorTemporal, Status: "started", WorkflowID: workflowID, WorkflowRunID: run.GetRunID()}, nil
}

func (s *Service) StartTranslation(ctx context.Context, req StartTranslationRequest) (StartResult, error) {
	if s.temporal == nil {
		return StartResult{}, errors.New("temporal client not available")
	}
	workflowID := translationWorkflowID(req.Source, len(req.IDs) == 0 && len(req.Filters) == 0, s.now())
	input := map[string]any{
		"source":         req.Source,
		"table_name":     req.TableName,
		"source_lang":    req.SourceLang,
		"target_lang":    "en",
		"ids":            req.IDs,
		"filters":        req.Filters,
		"prompt_version": envWithDefault("LLM_PROMPT_VERSION", "v1"),
		"model":          envWithDefault("LLM_MODEL", "qwen3:6b"),
		"fx_rate_date":   req.FXRateDate,
	}
	if req.Source == "brreg" {
		delete(input, "source")
		delete(input, "table_name")
		delete(input, "source_lang")
		delete(input, "target_lang")
	}
	run, err := s.temporal.ExecuteWorkflow(ctx, client.StartWorkflowOptions{
		ID:        workflowID,
		TaskQueue: TaskQueue,
		WorkflowIDReusePolicy: enumspb.WORKFLOW_ID_REUSE_POLICY_ALLOW_DUPLICATE,
		WorkflowExecutionErrorWhenAlreadyStarted: true,
		SearchAttributes: searchAttributes(req.Source, "translate"),
		Memo: memo(req.Source, "translate", req.Trigger, map[string]any{
			"ids":          req.IDs,
			"filters":      req.Filters,
			"fx_rate_date": req.FXRateDate,
		}),
	}, req.WorkflowType, input)
	if err != nil {
		return StartResult{}, errors.Wrap(err, "start temporal translation workflow")
	}
	return StartResult{Executor: ExecutorTemporal, Status: "started", WorkflowID: workflowID, WorkflowRunID: run.GetRunID()}, nil
}

func sourceWorkflowID(source, country string, now time.Time) string {
	region := country
	if region == "" {
		region = "global"
	}
	return fmt.Sprintf("pull-%s-%s-%s", source, region, now.UTC().Format("20060102150405"))
}

func translationWorkflowID(source string, singleton bool, now time.Time) string {
	if singleton {
		return fmt.Sprintf("translate-%s-all", source)
	}
	return fmt.Sprintf("translate-%s-%s", source, now.UTC().Format("20060102150405"))
}

func envWithDefault(key, fallback string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return fallback
}

func (s *Service) ListTasks(ctx context.Context, filter TaskFilter) TaskPage {
	if filter.Limit <= 0 || filter.Limit > 200 {
		filter.Limit = 50
	}
	var page TaskPage
	if filter.Executor == "" || filter.Executor == ExecutorRiver {
		rows, err := s.db.ListRiverTasks(ctx, db.ListRiverTasksParams{
			Source: stringPtrOrNil(filter.Source),
			Action: stringPtrOrNil(filter.Action),
			Status: stringPtrOrNil(filter.Status),
			Kind:   stringPtrOrNil(filter.Kind),
			Before: filter.Before,
			Limit:  filter.Limit,
		})
		if err != nil {
			page.Partial = true
			page.Errors = append(page.Errors, TaskError{Executor: ExecutorRiver, Message: "failed to load River tasks"})
		} else {
			for _, row := range rows {
				page.Items = append(page.Items, riverTaskFromRow(row))
			}
		}
	}
	if filter.Executor == "" || filter.Executor == ExecutorTemporal {
		if s.temporal == nil {
			page.Partial = true
			page.Errors = append(page.Errors, TaskError{Executor: ExecutorTemporal, Message: "temporal client not available"})
		} else {
			resp, err := s.temporal.ListWorkflow(ctx, &workflowservice.ListWorkflowExecutionsRequest{
				Namespace: Namespace,
				PageSize:  filter.Limit,
				Query:     buildTemporalVisibilityQuery(filter),
			})
			if err != nil {
				page.Partial = true
				page.Errors = append(page.Errors, TaskError{Executor: ExecutorTemporal, Message: "failed to load Temporal tasks"})
			} else {
				for _, info := range resp.Executions {
					page.Items = append(page.Items, temporalTaskFromExecution(info, s.temporalUIURL))
				}
			}
		}
	}
	sort.Slice(page.Items, func(i, j int) bool {
		return page.Items[i].CreatedAt.After(page.Items[j].CreatedAt)
	})
	if int32(len(page.Items)) > filter.Limit {
		page.Items = page.Items[:filter.Limit]
	}
	if len(page.Items) > 0 {
		next := page.Items[len(page.Items)-1].CreatedAt
		page.NextBefore = &next
	}
	return page
}

func (s *Service) TaskStats(ctx context.Context, filter TaskFilter) TaskStatsResponse {

func (s *Service) temporalStats(ctx context.Context, filter TaskFilter, resp *TaskStatsResponse) []TaskStatsItem {
	if s.temporal == nil {
		resp.Partial = true
		resp.Errors = append(resp.Errors, TaskError{Executor: ExecutorTemporal, Message: "temporal client not available"})
		return nil
	}
	sources := valuesOrKnown(filter.Source, knownTemporalSources)
	actions := valuesOrKnown(filter.Action, knownTemporalActions)
	statuses := valuesOrKnown(filter.Status, []string{"queued", "running", "completed", "failed", "cancelled"})
	items := make([]TaskStatsItem, 0, len(sources)*len(actions)*len(statuses))
	for _, source := range sources {
		for _, action := range actions {
			for _, status := range statuses {
				countFilter := filter
				countFilter.Source = source
				countFilter.Action = action
				countFilter.Status = status
				count, err := s.temporal.CountWorkflow(ctx, &workflowservice.CountWorkflowExecutionsRequest{
					Namespace: Namespace,
					Query:     buildTemporalVisibilityQuery(countFilter),
				})
				if err != nil {
					resp.Partial = true
					resp.Errors = append(resp.Errors, TaskError{Executor: ExecutorTemporal, Message: "failed to load Temporal task stats"})
					return items
				}
				if count.Count > 0 {
					items = append(items, TaskStatsItem{Executor: ExecutorTemporal, Source: source, Action: action, Status: status, Count: count.Count})
				}
			}
		}
	}
	return items
}

func valuesOrKnown(value string, known []string) []string {
	if value != "" {
		return []string{value}
	}
	return known
}

import (

func (h *Handlers) handleListTasks(w http.ResponseWriter, r *http.Request) {
	if h.taskService == nil {
		writeError(w, http.StatusServiceUnavailable, "task service not available")
		return
	}
	filter, ok := taskFilterFromRequest(w, r)
	if !ok {
		return
	}
	writeJSON(w, http.StatusOK, h.taskService.ListTasks(r.Context(), filter))
}

func (h *Handlers) handleTaskStats(w http.ResponseWriter, r *http.Request) {
	if h.taskService == nil {
		writeError(w, http.StatusServiceUnavailable, "task service not available")
		return
	}
	filter, ok := taskFilterFromRequest(w, r)
	if !ok {
		return
	}
	writeJSON(w, http.StatusOK, h.taskService.TaskStats(r.Context(), filter))
}

func taskFilterFromRequest(w http.ResponseWriter, r *http.Request) (tasksvc.TaskFilter, bool) {
	limit := min(queryInt(r, "limit", 50), 200)
	filter := tasksvc.TaskFilter{
		Executor: r.URL.Query().Get("executor"),
		Source:   r.URL.Query().Get("source"),
		Action:   r.URL.Query().Get("action"),
		Status:   r.URL.Query().Get("status"),
		Kind:     r.URL.Query().Get("kind"),
		Limit:    int32(limit),
	}
	if before := r.URL.Query().Get("before"); before != "" {
		parsed, err := time.Parse(time.RFC3339, before)
		if err != nil {
			writeError(w, http.StatusBadRequest, "invalid before cursor")
			return tasksvc.TaskFilter{}, false
		}
		filter.Before = &parsed
	}
	return filter, true
}
