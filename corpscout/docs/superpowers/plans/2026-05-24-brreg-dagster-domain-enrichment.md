# BRREG Dagster Domain Enrichment Implementation Plan

Status: Superseded by `docs/superpowers/specs/2026-05-24-brreg-enhanced-source-handoff-design.md`.

This plan follows the earlier domain-only split. Do not implement it as written. The current direction is to first analyze BRREG raw, translated, and financial payloads, add a versioned enhanced artifact table, then let Corpscout unpack that artifact into normalized BRREG source tables.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first BRREG enrichment vertical slice: raw BRREG inputs are normalized into source tables, Dagster orchestrates domain enrichment through Temporal, and Corpscout reads the normalized/enriched BRREG source records.

**Architecture:** Corpscout Postgres remains the data store and review system. Dagster owns the BRREG enrichment graph and writes normalized/enriched source tables. Temporal remains a bounded durable worker for domain discovery and returns discovery results to Dagster for the MVP.

**Tech Stack:** PostgreSQL migrations, sqlc, Go scheduler API/tests, React Router UI, Python Dagster, psycopg, Temporal Python client, existing data-pipelines Go Temporal workflow.

CREATE TABLE brreg_source_companies (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  raw_input_id UUID NOT NULL REFERENCES brreg_company_raw_inputs(id) ON DELETE CASCADE,
  organization_number TEXT NOT NULL,
  name TEXT,
  registration_status TEXT,
  country_iso2 TEXT NOT NULL DEFAULT 'NO',
  payload_hash TEXT NOT NULL,
  source_updated_at TIMESTAMPTZ,
  normalized_payload JSONB NOT NULL DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT chk_brreg_source_companies_payload_object CHECK (jsonb_typeof(normalized_payload) = 'object'),
  UNIQUE (organization_number, payload_hash)
);

CREATE INDEX idx_brreg_source_companies_raw_input
  ON brreg_source_companies(raw_input_id);
CREATE INDEX idx_brreg_source_companies_org
  ON brreg_source_companies(organization_number);
CREATE INDEX idx_brreg_source_companies_name
  ON brreg_source_companies(name);

CREATE TABLE brreg_enrichment_runs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  dagster_run_id TEXT,
  asset_key TEXT NOT NULL,
  step TEXT NOT NULL,
  scope JSONB NOT NULL DEFAULT '{}'::jsonb,
  status TEXT NOT NULL,
  started_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  finished_at TIMESTAMPTZ,
  error TEXT,
  metadata JSONB NOT NULL DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT chk_brreg_enrichment_runs_status CHECK (
    status IN ('queued', 'running', 'succeeded', 'failed', 'cancelled')
  ),
  CONSTRAINT chk_brreg_enrichment_runs_scope_object CHECK (jsonb_typeof(scope) = 'object'),
  CONSTRAINT chk_brreg_enrichment_runs_metadata_object CHECK (jsonb_typeof(metadata) = 'object')
);

CREATE INDEX idx_brreg_enrichment_runs_step_status
  ON brreg_enrichment_runs(step, status, started_at DESC);
CREATE INDEX idx_brreg_enrichment_runs_dagster_run
  ON brreg_enrichment_runs(dagster_run_id)
  WHERE dagster_run_id IS NOT NULL;

CREATE TABLE brreg_source_domain_observations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_company_id UUID NOT NULL REFERENCES brreg_source_companies(id) ON DELETE CASCADE,
  raw_input_id UUID NOT NULL REFERENCES brreg_company_raw_inputs(id) ON DELETE CASCADE,
  enrichment_run_id UUID REFERENCES brreg_enrichment_runs(id) ON DELETE SET NULL,
  domain TEXT NOT NULL,
  signal TEXT NOT NULL,
  confidence SMALLINT NOT NULL,
  evidence JSONB NOT NULL DEFAULT '{}'::jsonb,
  metadata JSONB NOT NULL DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT chk_brreg_source_domain_observations_confidence CHECK (confidence BETWEEN 1 AND 100),
  CONSTRAINT chk_brreg_source_domain_observations_evidence_object CHECK (jsonb_typeof(evidence) = 'object'),
  CONSTRAINT chk_brreg_source_domain_observations_metadata_object CHECK (jsonb_typeof(metadata) = 'object')
);

CREATE INDEX idx_brreg_source_domain_observations_company
  ON brreg_source_domain_observations(source_company_id, created_at DESC);
CREATE INDEX idx_brreg_source_domain_observations_run
  ON brreg_source_domain_observations(enrichment_run_id)
  WHERE enrichment_run_id IS NOT NULL;

CREATE TABLE brreg_source_company_domains (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_company_id UUID NOT NULL REFERENCES brreg_source_companies(id) ON DELETE CASCADE,
  domain TEXT NOT NULL,
  best_signal TEXT NOT NULL,
  confidence SMALLINT NOT NULL,
  status TEXT NOT NULL DEFAULT 'active',
  evidence JSONB NOT NULL DEFAULT '{}'::jsonb,
  first_seen_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  last_seen_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT chk_brreg_source_company_domains_confidence CHECK (confidence BETWEEN 1 AND 100),
  CONSTRAINT chk_brreg_source_company_domains_status CHECK (status IN ('active', 'superseded', 'rejected')),
  CONSTRAINT chk_brreg_source_company_domains_evidence_object CHECK (jsonb_typeof(evidence) = 'object'),
  UNIQUE (source_company_id, domain)
);

CREATE INDEX idx_brreg_source_company_domains_company_status
  ON brreg_source_company_domains(source_company_id, status);
CREATE INDEX idx_brreg_source_company_domains_domain
  ON brreg_source_company_domains(domain);

GRANT SELECT ON brreg_source_companies TO corpscout_anon;
GRANT SELECT ON brreg_enrichment_runs TO corpscout_anon;

```sql
-- name: ListBrregSourceCompanies :many
SELECT
  c.id,
  c.raw_input_id,
  c.organization_number,
  COALESCE(c.name, '')::text AS name,
  COALESCE(c.registration_status, '')::text AS registration_status,
  c.country_iso2,
  c.payload_hash,
  c.source_updated_at,
  c.normalized_payload,
  c.created_at,
  c.updated_at,
  COALESCE(domains.connected_domain_count, 0)::bigint AS connected_domain_count,
  COALESCE(latest_run.status, 'not_started')::text AS latest_domain_enrichment_status
FROM brreg_source_companies c
LEFT JOIN LATERAL (
  SELECT count(*)::bigint AS connected_domain_count
  FROM brreg_source_company_domains d
  WHERE d.source_company_id = c.id
    AND d.status = 'active'
) domains ON true
LEFT JOIN LATERAL (
  SELECT r.status
  FROM brreg_enrichment_runs r
  WHERE r.step = 'domain_enrichment'
    AND r.scope->>'source_company_id' = c.id::text
  ORDER BY r.started_at DESC
  LIMIT 1
) latest_run ON true
WHERE (sqlc.narg('q')::text IS NULL
       OR c.name ILIKE '%' || sqlc.narg('q')::text || '%'
       OR c.organization_number ILIKE '%' || sqlc.narg('q')::text || '%')
ORDER BY c.created_at DESC
LIMIT sqlc.arg('limit') OFFSET sqlc.arg('offset_rows');

-- name: CountBrregSourceCompanies :one
SELECT count(*)::bigint
FROM brreg_source_companies c
WHERE (sqlc.narg('q')::text IS NULL
       OR c.name ILIKE '%' || sqlc.narg('q')::text || '%'
       OR c.organization_number ILIKE '%' || sqlc.narg('q')::text || '%');

-- name: GetBrregSourceCompany :one
SELECT
  c.id,
  c.raw_input_id,
  c.organization_number,
  COALESCE(c.name, '')::text AS name,
  COALESCE(c.registration_status, '')::text AS registration_status,
  c.country_iso2,
  c.payload_hash,
  c.source_updated_at,
  c.normalized_payload,
  c.created_at,
  c.updated_at
FROM brreg_source_companies c
WHERE c.id = $1;

-- name: GetBrregSourceCompanyByRawInput :one
SELECT
  c.id,
  c.raw_input_id,
  c.organization_number,
  COALESCE(c.name, '')::text AS name,
  COALESCE(c.registration_status, '')::text AS registration_status,
  c.country_iso2,
  c.payload_hash,
  c.source_updated_at,
  c.normalized_payload,
  c.created_at,
  c.updated_at
FROM brreg_source_companies c
WHERE c.raw_input_id = $1;

-- name: ListBrregSourceCompanyDomains :many
SELECT id, source_company_id, domain, best_signal, confidence, status, evidence, first_seen_at, last_seen_at
FROM brreg_source_company_domains
WHERE source_company_id = $1
ORDER BY confidence DESC, domain ASC;
```

package httpapi

import (
	"log/slog"
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/google/uuid"

	db "github.com/pulsarpoint/corpscout/scheduler/internal/db/gen"
)

type brregSourceCompanyListResponse struct {
	Items []db.ListBrregSourceCompaniesRow `json:"items"`
	Page  int                              `json:"page"`
	Limit int                              `json:"limit"`
	Total int64                            `json:"total"`
}

type brregSourceCompanyDetailResponse struct {
	Company db.GetBrregSourceCompanyRow   `json:"company"`
	Domains []db.BrregSourceCompanyDomain `json:"domains"`
}

func (h *Handlers) handleListBrregSourceCompanies(w http.ResponseWriter, r *http.Request) {
	if h.pool == nil {
		writeError(w, http.StatusServiceUnavailable, "database pool not available")
		return
	}
	page := queryInt(r, "page", 1)
	limit := queryInt(r, "limit", 50)
	if limit > 200 {
		limit = 200
	}
	q := queryString(r, "q")
	queries := db.New(h.pool)

	total, err := queries.CountBrregSourceCompanies(r.Context(), q)
	if err != nil {
		slog.Error("count brreg source companies", "error", err)
		writeError(w, http.StatusInternalServerError, "internal error")
		return
	}
	items, err := queries.ListBrregSourceCompanies(r.Context(), db.ListBrregSourceCompaniesParams{
		Q:          q,
		Limit:      int32(limit),
		OffsetRows: int32((page - 1) * limit),
	})
	if err != nil {
		slog.Error("list brreg source companies", "error", err)
		writeError(w, http.StatusInternalServerError, "internal error")
		return
	}
	writeJSON(w, http.StatusOK, brregSourceCompanyListResponse{Items: items, Page: page, Limit: limit, Total: total})
}

func (h *Handlers) handleGetBrregSourceCompany(w http.ResponseWriter, r *http.Request) {
	if h.pool == nil {
		writeError(w, http.StatusServiceUnavailable, "database pool not available")
		return
	}
	id, err := uuid.Parse(chi.URLParam(r, "id"))
	if err != nil {
		writeError(w, http.StatusBadRequest, "invalid source company id")
		return
	}
	queries := db.New(h.pool)
	company, err := queries.GetBrregSourceCompany(r.Context(), id)
	if err != nil {
		writeError(w, http.StatusNotFound, "source company not found")
		return
	}
	domains, err := queries.ListBrregSourceCompanyDomains(r.Context(), id)
	if err != nil {
		slog.Error("list brreg source company domains", "source_company_id", id, "error", err)
		writeError(w, http.StatusInternalServerError, "internal error")
		return
	}
	writeJSON(w, http.StatusOK, brregSourceCompanyDetailResponse{Company: company, Domains: domains})
}

def rebuild_brreg_source_companies(database_url: str) -> int:

        with conn.cursor() as cur:

            cur.execute("""
                SELECT id::text, organization_number, payload_hash, raw_payload, source_updated_at

                payload = raw_payload if isinstance(raw_payload, dict) else json.loads(raw_payload)
                normalized = normalize_brreg_payload(payload)

                        raw_input_id, organization_number, name, registration_status,
                        country_iso2, payload_hash, source_updated_at, normalized_payload

                    VALUES (%s::uuid, %s, %s, %s, %s, %s, %s, %s::jsonb)

                        registration_status = EXCLUDED.registration_status,
                        country_iso2 = EXCLUDED.country_iso2,

                    normalized["country_iso2"],
                    payload_hash,
                    source_updated_at,

from urllib.parse import urlparse


def normalize_domain(value: str) -> str:
    raw = (value or "").strip().lower()
    if not raw:
        return ""
    parsed = urlparse(raw if "://" in raw else f"https://{raw}")
    host = parsed.hostname or raw.split("/", 1)[0]
    return host.rstrip(".")


def choose_best_domain_observations(observations: list[dict]) -> list[dict]:
    best_by_domain: dict[str, dict] = {}
    for observation in observations:
        domain = normalize_domain(str(observation.get("domain") or ""))
        if not domain:
            continue
        candidate = {
            "domain": domain,
            "best_signal": str(observation.get("signal") or "search"),
            "confidence": int(observation.get("confidence") or 1),
            "evidence": observation.get("evidence") if isinstance(observation.get("evidence"), dict) else {},
        }
        current = best_by_domain.get(domain)
        if current is None or candidate["confidence"] > current["confidence"]:
            best_by_domain[domain] = candidate
    return sorted(best_by_domain.values(), key=lambda item: item["domain"])
```

                        last_seen_at = now()
                """, (source_company_id, item["domain"], item["best_signal"], item["confidence"], json.dumps(item["evidence"])))
            cur.execute("""
                UPDATE brreg_enrichment_runs
                SET status = 'succeeded', finished_at = now(), updated_at = now(),
                    metadata = metadata || jsonb_build_object('domain_count', %s)
                WHERE id = %s::uuid
            """, (len(best), enrichment_run_id))
            conn.commit()
            return len(best)
```
